cmake_minimum_required(VERSION 3.20)
project(FotoVLC VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_library(decoders src/decoders.cpp)
target_include_directories(decoders PUBLIC src)
add_executable(decoder_test tests/decoder_test.cpp)
target_link_libraries(decoder_test PRIVATE decoders)
add_test(NAME decoders COMMAND decoder_test)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE src)
add_test(NAME core COMMAND core_test)
if(WIN32)
    enable_language(RC)
    add_executable(FotoVLC WIN32 src/main.cpp assets/app.rc)
    target_include_directories(FotoVLC PRIVATE assets)
    if(MSVC)
        target_compile_options(FotoVLC PRIVATE /utf-8 /W4)
        # app.rc embeds our DPI-aware, asInvoker manifest; avoid a duplicate resource.
        target_link_options(FotoVLC PRIVATE /MANIFEST:NO)
        target_compile_options(core_test PRIVATE /utf-8 /W4)
    endif()
    target_compile_definitions(FotoVLC PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(FotoVLC PRIVATE decoders windowscodecs gdiplus ole32 shell32 comdlg32 shlwapi dwmapi uuid)
    if(MINGW)
        target_link_options(FotoVLC PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS FotoVLC RUNTIME DESTINATION .)
endif()
