cmake_minimum_required(VERSION 3.20)
project(PhotoVLC VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(BUILD_SHARED_LIBS OFF)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(third_party/libwebp EXCLUDE_FROM_ALL)
add_library(decoders src/decoders.cpp)
target_link_libraries(decoders PRIVATE webpdemux)
target_include_directories(decoders PUBLIC src)
add_executable(decoder_test tests/decoder_test.cpp)
target_link_libraries(decoder_test PRIVATE decoders)
add_test(NAME decoders COMMAND decoder_test)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE src)
add_test(NAME core COMMAND core_test)
if(WIN32)
    enable_language(RC)
    configure_file(src/version.h.in generated/version.h @ONLY)
    configure_file(assets/app.manifest.in generated/app.manifest @ONLY)
    add_executable(PhotoVLC WIN32 src/main.cpp src/image.cpp assets/app.rc)
    target_include_directories(PhotoVLC PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" assets)
    if(MSVC)
        target_compile_options(PhotoVLC PRIVATE /utf-8 /W4)
        # app.rc embeds our DPI-aware, asInvoker manifest; avoid a duplicate resource.
        target_link_options(PhotoVLC PRIVATE /MANIFEST:NO)
        target_compile_options(core_test PRIVATE /utf-8 /W4)
    endif()
    target_compile_definitions(PhotoVLC PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PhotoVLC PRIVATE decoders windowscodecs gdiplus ole32 shell32 comdlg32 shlwapi dwmapi uuid comctl32 version advapi32)
    if(MINGW)
        target_link_options(PhotoVLC PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS PhotoVLC RUNTIME DESTINATION .)
endif()
