cmake_minimum_required(VERSION 3.20)
project(PAFPhotos VERSION 0.7.4 LANGUAGES C CXX)
set(PAF_RELEASE_VERSION "0.7.4")
set(PAF_VERSION_REVISION 0)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
set(BUILD_SHARED_LIBS OFF)
foreach(option ANIM_UTILS CWEBP DWEBP GIF2WEBP IMG2WEBP VWEBP WEBPINFO LIBWEBPMUX WEBPMUX EXTRAS)
    set(WEBP_BUILD_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory(third_party/libwebp EXCLUDE_FROM_ALL)
add_library(decoders src/decoders.cpp src/extra_decoders.cpp)
target_link_libraries(decoders PRIVATE webpdemux)
target_include_directories(decoders PUBLIC src)
add_executable(decoder_test tests/decoder_test.cpp)
target_link_libraries(decoder_test PRIVATE decoders)
add_test(NAME decoders COMMAND decoder_test)
add_executable(core_test tests/core_test.cpp)
target_include_directories(core_test PRIVATE src)
add_test(NAME core COMMAND core_test)
add_executable(update_test tests/update_test.cpp)
target_include_directories(update_test PRIVATE src)
add_test(NAME updates COMMAND update_test)
add_executable(motion_test tests/motion_test.cpp)
target_include_directories(motion_test PRIVATE src)
add_test(NAME motion COMMAND motion_test)
add_executable(edit_test tests/edit_test.cpp)
target_include_directories(edit_test PRIVATE src)
add_test(NAME edits COMMAND edit_test)
add_executable(editor_engine_test tests/editor_engine_test.cpp)
target_include_directories(editor_engine_test PRIVATE src)
add_test(NAME editor_engine COMMAND editor_engine_test)
if(WIN32)
    enable_language(RC)
    configure_file(src/version.h.in generated/version.h @ONLY)
    configure_file(assets/app.manifest.in generated/app.manifest @ONLY)
    set_property(SOURCE assets/app.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/generated/version.h;${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest")
    add_executable(PAFPhotos WIN32 src/main.cpp src/image.cpp src/export.cpp src/updater.cpp src/codec_pack.cpp src/resize_dialog.cpp src/photo_editor.cpp src/decoder_catalog.cpp assets/app.rc)
    add_executable(PAFPhotosUpdater WIN32 src/update_helper.cpp src/updater.cpp)
    set_target_properties(PAFPhotosUpdater PROPERTIES OUTPUT_NAME "PAFPhotos.Update")
    target_compile_definitions(PAFPhotosUpdater PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PAFPhotosUpdater PRIVATE winhttp bcrypt advapi32 shell32)
    target_include_directories(PAFPhotos PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" assets)
    if(MSVC)
        target_compile_options(PAFPhotosUpdater PRIVATE /utf-8 /W4)
        target_compile_options(PAFPhotos PRIVATE /utf-8 /W4)
        # app.rc embeds our DPI-aware, asInvoker manifest; avoid a duplicate resource.
        target_link_options(PAFPhotos PRIVATE /MANIFEST:NO)
        target_compile_options(core_test PRIVATE /utf-8 /W4)
    endif()
    target_compile_definitions(PAFPhotos PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(PAFPhotos PRIVATE decoders webp windowscodecs gdiplus ole32 shell32 comdlg32 shlwapi dwmapi uxtheme uuid comctl32 version advapi32 winhttp bcrypt)
    if(MINGW)
        target_link_options(PAFPhotosUpdater PRIVATE -municode -static -static-libgcc -static-libstdc++)
        target_link_options(PAFPhotos PRIVATE -municode -static -static-libgcc -static-libstdc++)
    endif()
    install(TARGETS PAFPhotos RUNTIME DESTINATION .)
endif()

option(PAF_BUILD_LINUX "Build the native Linux desktop application" OFF)
if(PAF_BUILD_LINUX AND UNIX AND NOT APPLE)
    find_package(Qt6 REQUIRED COMPONENTS Widgets Concurrent Network)
    find_package(Qt6Svg QUIET)
    find_path(PAF_JXL_INCLUDE jxl/decode.h REQUIRED)
    find_library(PAF_JXL_LIBRARY NAMES jxl REQUIRED)
    find_library(PAF_JXL_THREADS NAMES jxl_threads REQUIRED)
    qt_add_executable(PAFPhotosJXL src/linux/jxl.cpp)
    target_include_directories(PAFPhotosJXL PRIVATE ${PAF_JXL_INCLUDE})
    target_link_libraries(PAFPhotosJXL PRIVATE Qt6::Gui ${PAF_JXL_LIBRARY} ${PAF_JXL_THREADS})
    target_link_options(PAFPhotosJXL PRIVATE -Wl,--disable-new-dtags)
    set_target_properties(PAFPhotosJXL PROPERTIES OUTPUT_NAME PAFPhotos.JXL BUILD_RPATH "$ORIGIN/../lib/paf-photos" INSTALL_RPATH "$ORIGIN/../lib/paf-photos")
    install(TARGETS PAFPhotosJXL RUNTIME DESTINATION bin)
    qt_add_executable(PAFPhotosLinux src/linux/main.cpp)
    qt_add_resources(PAFPhotosLinux "paf_resources" FILES assets/logo.png PREFIX "/paf" BASE assets)
    target_include_directories(PAFPhotosLinux PRIVATE src)
    target_compile_definitions(PAFPhotosLinux PRIVATE PAF_RELEASE_VERSION="${PAF_RELEASE_VERSION}")
    target_link_libraries(PAFPhotosLinux PRIVATE decoders Qt6::Widgets Qt6::Concurrent Qt6::Network)
    set_target_properties(PAFPhotosLinux PROPERTIES OUTPUT_NAME PAFPhotos)
    install(TARGETS PAFPhotosLinux RUNTIME DESTINATION bin)
    install(FILES packaging/linux/paf-photos.desktop DESTINATION share/applications)
    install(FILES assets/logo.png DESTINATION share/icons/hicolor/256x256/apps RENAME paf-photos.png)
endif()
